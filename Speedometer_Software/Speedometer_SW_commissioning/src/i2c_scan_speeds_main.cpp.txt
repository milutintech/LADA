// src/main.cpp
#include <Arduino.h>
#include <Wire.h>
#include <driver/i2c.h>

#define SDA_PIN 1
#define SCL_PIN 2

// I2C speed modes
#define I2C_SPEED_LOW      10000   // 10 kHz - Super slow for troubleshooting
#define I2C_SPEED_STD      100000  // 100 kHz - Standard mode
#define I2C_SPEED_FM       400000  // 400 kHz - Fast mode
#define I2C_SPEED_FMP      1000000 // 1 MHz - Fast mode plus

struct I2CError {
    uint32_t timeoutCount = 0;
    uint32_t nackCount = 0;
    uint32_t arbitrationLostCount = 0;
    uint32_t unknownErrorCount = 0;
    uint32_t totalTransactions = 0;
};

I2CError errorStats;

// Reinitialize I2C with new speed
void reinitI2C(uint32_t speed) {
    Wire.end();
    Wire.begin(SDA_PIN, SCL_PIN);
    Wire.setClock(speed);
    delay(100); // Give devices time to adjust
    
    Serial.printf("\nI2C Speed set to %d Hz\n", speed);
}

bool isI2CBusStuck() {
    bool sdaState = digitalRead(SDA_PIN);
    bool sclState = digitalRead(SCL_PIN);
    
    if (!sdaState || !sclState) {
        Serial.println("WARNING: I2C bus might be stuck!");
        Serial.printf("SDA State: %d, SCL State: %d\n", sdaState, sclState);
        return true;
    }
    return false;
}

void recoverI2CBus() {
    Serial.println("Attempting I2C bus recovery...");
    Wire.end();
    
    pinMode(SDA_PIN, OUTPUT);
    pinMode(SCL_PIN, OUTPUT);
    
    digitalWrite(SDA_PIN, HIGH);
    for(int i = 0; i < 9; i++) {
        digitalWrite(SCL_PIN, HIGH);
        delayMicroseconds(20); // Longer delay for slower devices
        digitalWrite(SCL_PIN, LOW);
        delayMicroseconds(20);
    }
    
    digitalWrite(SDA_PIN, LOW);
    delayMicroseconds(20);
    digitalWrite(SCL_PIN, HIGH);
    delayMicroseconds(20);
    digitalWrite(SDA_PIN, HIGH);
    delayMicroseconds(20);
    
    Wire.begin(SDA_PIN, SCL_PIN);
}

void analyzeError(uint8_t error, uint8_t address) {
    errorStats.totalTransactions++;
    
    switch(error) {
        case 0: return;
        case 2: // NACK on address
            Serial.printf("Error at 0x%02X: NACK on address\n", address);
            errorStats.nackCount++;
            break;
        case 3: // NACK on data
            Serial.printf("Error at 0x%02X: NACK on data\n", address);
            errorStats.nackCount++;
            break;
        case 4: // Other error
            Serial.printf("Error at 0x%02X: Other error\n", address);
            errorStats.unknownErrorCount++;
            break;
        case 5: // Timeout
            Serial.printf("Error at 0x%02X: Timeout\n", address);
            errorStats.timeoutCount++;
            break;
        case 6: // Arbitration lost
            Serial.printf("Error at 0x%02X: Arbitration lost (collision)\n", address);
            errorStats.arbitrationLostCount++;
            break;
    }
}

void scanI2CBusAtSpeed(uint32_t speed) {
    reinitI2C(speed);
    
    byte error, address;
    int devicesFound = 0;
    
    // Reset error statistics
    memset(&errorStats, 0, sizeof(errorStats));
    
    Serial.printf("\nScanning I2C bus at %d Hz...\n", speed);
    
    // Check for stuck bus
    if (isI2CBusStuck()) {
        recoverI2CBus();
        reinitI2C(speed); // Reinitialize at the same speed after recovery
    }
    
    // Scan all addresses with multiple attempts
    for (address = 1; address < 128; address++) {
        bool deviceFound = false;
        
        // Try multiple times to detect intermittent issues
        for (int attempt = 0; attempt < 3; attempt++) {
            Wire.beginTransmission(address);
            error = Wire.endTransmission();
            
            analyzeError(error, address);
            
            if (error == 0) {
                if (!deviceFound) {
                    Serial.printf("Device found at address 0x%02X\n", address);
                    devicesFound++;
                    deviceFound = true;
                }
            }
            delay(5); // Longer delay between attempts
        }
    }
    
    // Print results
    Serial.printf("\nI2C Bus Scan Results at %d Hz:\n", speed);
    Serial.printf("Devices found: %d\n", devicesFound);
    Serial.printf("Total transactions: %u\n", errorStats.totalTransactions);
    Serial.printf("NACK errors: %u (%.1f%%)\n", 
                 errorStats.nackCount,
                 (float)errorStats.nackCount * 100 / errorStats.totalTransactions);
    Serial.printf("Arbitration lost: %u (%.1f%%)\n", 
                 errorStats.arbitrationLostCount,
                 (float)errorStats.arbitrationLostCount * 100 / errorStats.totalTransactions);
    Serial.printf("Timeouts: %u (%.1f%%)\n", 
                 errorStats.timeoutCount,
                 (float)errorStats.timeoutCount * 100 / errorStats.totalTransactions);
}

void setup() {
    Serial.begin(115200);
    delay(1000);
    Serial.println("\nI2C Multi-Speed Bus Scanner");
}

void loop() {
    // Test at different speeds, from slowest to fastest
    scanI2CBusAtSpeed(I2C_SPEED_LOW);   // 10 kHz
    delay(1000);
    
    scanI2CBusAtSpeed(I2C_SPEED_STD);   // 100 kHz
    delay(1000);
    
    scanI2CBusAtSpeed(I2C_SPEED_FM);    // 400 kHz
    delay(1000);
    
    Serial.println("\n--- Scan complete, waiting 10 seconds before next scan ---\n");
    delay(10000);
}
